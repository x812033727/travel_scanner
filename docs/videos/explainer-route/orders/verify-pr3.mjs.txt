// 2. The table, the names, the new keys.
assert.deepEqual(Object.keys(own.EXPLAINER_VARIANTS), [...drama.EXPLAINER_PRESETS]);
assert.deepEqual(own.EXPLAINER_VARIANTS["flat-explainer"], { bible: "bible-explainer", stage: "explainer", discuss: "discuss" });
assert.deepEqual(Object.keys(own.EXPLAINER_SERIES_PROFILES), [...drama.CUTE_EXPLAINER_PRESETS]);
const sent = (variants) => [["planner", variants.bible], ["writer", variants.stage], ["verifier", variants.stage], ["planner", variants.discuss], ["writer", variants.discuss]];
for (const [preset, variants] of Object.entries(own.EXPLAINER_VARIANTS)) {
  for (const [stage, variant] of sent(variants)) {
    assert.match(variant, /^[a-z][a-z0-9_-]{0,31}$/);
    assert.ok(`${stage}/${variant}`.length <= 32, `${stage}/${variant}`);
    if (preset === "flat-explainer") continue;
    const key = `${stage}:${variant}`;
    assert.equal(Object.hasOwn(prompts.VARIANT_INSTRUCTIONS, key), false, `${key} collides`);
    const text = instructionsFor(stage, "drama", "", variant);
    assert.equal(text, own.EXPLAINER_SERIES_INSTRUCTIONS[key], key);
    assert.equal(instructionsFor(stage, "drama", "", variant, "", row), text, `${key} takes no genre section`);
    assert.ok(text.length < 40_000, key);
    const profile = own.EXPLAINER_SERIES_PROFILES[preset];
    assert.ok(text.includes(profile.series) && text.includes(profile.docs), key);
    for (const phrase of ["flat editorial", "burned-in", '"preset": "flat-explainer"', "No mascot or recurring character", "burn_in: true", "anonymous figures"]) assert.equal(text.includes(phrase), false, `${key} has ${phrase}`);
    assert.equal(text.includes(preset === "cute-mystery" ? "原來如此" : "奇聞檔案局"), false, key);
    if (stage === "writer") {
      assert.ok(text.includes(`"look": {"preset": "${preset}"}`), key);
      for (const card of drama.explainerCardTemplates(preset)) assert.ok(text.includes(`"${card}" {`), `${key} names ${card}`);
      assert.match(text, /ONE person doing ONE concrete thing/);
      assert.match(text, /Never a still life or an empty scene on its own/);
      assert.match(text, /at least a third of the lines carry one \(lint warns below\s+that\)/);
    }
  }
}

assert.equal(/pillar: business/.test(mysteryWriter), false);
assert.match(instructionsFor("writer", "drama", "", "explainer-cartoon"), /pillar: business\|science\|travel\|tech/);
assert.match(instructionsFor("planner", "drama", "", "bible-explainer-mystery"), /「流傳的版本：…」 once, first[\s\S]*「紀錄：…」 two to four times[\s\S]*「目前最站得住的解釋：…」 once, last/);
assert.equal(instructionsFor("listener", "drama", "", "explainer-mystery"), instructionsFor("listener", "drama"), "the listener has no text of its own");
console.log("2 ok: table, names, lengths and the ten new texts");

const table = { id: "names", template: "table", data: { title: "這個地方的名字", columns: ["名字", "時間"], rows: [["地圖編號", "一九五五"], ["夢境", "六〇年代末"]] }, lines: [{ id: "t1ex", text: "第一個名字是地圖編號。", reveal: 1 }, { id: "t2ex", text: "第二個是無線電呼號。", reveal: 1 }] };

assert.equal(thumbnailSeries(mystery), null);
assert.deepEqual(renderProblems(mystery), []);
assert.deepEqual(thumbnailHeadlineProblems(mystery), []);
const html = thumbnailHtml(mystery.thumbnail, { series: thumbnailSeries(mystery), slug: mystery.slug });
assert.equal(html.includes("原來如此事務所") || html.includes("thumb-stamp") || /pillar-/.test(html), false);
assert.match(renderProblems({ ...mystery, thumbnail: { ...mystery.thumbnail, data: { ...mystery.thumbnail.data, pillar: "H" } } })[0].message, /pillar must be one of business/);
assert.match(renderProblems({ ...mystery, thumbnail: { ...mystery.thumbnail, data: { ...mystery.thumbnail.data, layout: "right" } } })[0].message, /layout must be one of a, b, c/);

// 7. The picture-variety rules hold a cute explainer and leave flat-explainer alone.
const shot = (id, camera, prompt = "Medium shot of a farm gate at dawn, a farmer in a waxed coat lifting the latch, low sun from the left") => ({ id, template: "shot", data: { prompt, camera, visual: "still" }, lines: [{ id: `${id}0x`.slice(0, 4), text: "農夫走到田邊。" }] });
const run = (preset, cameras, prompt) => drama.pictureVarietyProblems({ ...base, look: { preset }, scenes: cameras.map((camera, index) => shot(`p${index}`, camera, prompt)) });
assert.equal(run("cute-mystery", ["push in", "push in", "push in"]).errors.length, 1);
assert.match(run("cute-mystery", ["push in", "push in", "push in"]).errors[0].message, /3 stills in a row under "push in"/);
assert.equal(run("cute-cartoon", ["push in", "pull out", "push in"]).errors.length, 0);
assert.deepEqual(run("flat-explainer", ["push in", "push in", "push in"]), { errors: [], warnings: [] });
assert.match(run("cute-mystery", ["push in", "pull out"], "Medium shot of a chibi farmer at a gate at dawn, low sun from the left").warnings.map((w) => w.message).join(" | "), /restate the look \("chibi"\)/);
console.log("7 ok: picture variety");
